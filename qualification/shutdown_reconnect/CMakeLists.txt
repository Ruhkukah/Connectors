find_package(Threads REQUIRED)
# Separate libraries keep all injection controls and the hard no-post boundary
# out of normal moexctl / production archives.
foreach(pair IN ITEMS "moex_plaza2_trade_session" "moex_connector_host")
    get_target_property(original_sources ${pair} SOURCES)
    get_target_property(original_dir ${pair} SOURCE_DIR)
    list(TRANSFORM original_sources PREPEND "${original_dir}/")
    add_library(${pair}_shutdown_harness STATIC ${original_sources})
    get_target_property(original_includes ${pair} INTERFACE_INCLUDE_DIRECTORIES)
    target_include_directories(${pair}_shutdown_harness PUBLIC ${original_includes})
    get_target_property(original_links ${pair} INTERFACE_LINK_LIBRARIES)
    list(TRANSFORM original_links REPLACE "^moex_plaza2_trade_session$" "moex_plaza2_trade_session_shutdown_harness")
    target_link_libraries(${pair}_shutdown_harness PUBLIC ${original_links})
    target_compile_definitions(${pair}_shutdown_harness PUBLIC MOEX_ISOLATED_SHUTDOWN_HARNESS=1)
    target_compile_features(${pair}_shutdown_harness PUBLIC cxx_std_20)
endforeach()
add_executable(moex_shutdown_reconnect_harness main.cpp)
target_link_libraries(moex_shutdown_reconnect_harness PRIVATE moex_connector_host_shutdown_harness)
target_compile_definitions(moex_shutdown_reconnect_harness PRIVATE MOEX_SOURCE_GIT_SHA="${MOEX_SOURCE_GIT_SHA}")
set_target_properties(moex_shutdown_reconnect_harness PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${MOEX_BUILD_ROOT}/qualification")
# Actual CLI main/sigaction/drain loop, separately linked to the hard no-post session.
add_executable(moexctl_no_post "${MOEX_SOURCE_ROOT}/apps/moexctl.cpp")
target_link_libraries(moexctl_no_post PRIVATE moex_connector_host_shutdown_harness)
target_compile_definitions(moexctl_no_post PRIVATE MOEX_SOURCE_GIT_SHA="${MOEX_SOURCE_GIT_SHA}")
set_target_properties(moexctl_no_post PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${MOEX_BUILD_ROOT}/qualification")
