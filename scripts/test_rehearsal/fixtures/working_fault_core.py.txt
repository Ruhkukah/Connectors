class Refused(RuntimeError):

def require(value, reason):
    if not value:
        raise Refused(reason)


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def atomic(path, data):
    tmp = path.with_name(path.name + '.' + str(os.getpid()) + '.tmp')
    with tmp.open('x') as out:
        json.dump(data, out, sort_keys=True)
        out.write('\n')
        out.flush()
        os.fsync(out.fileno())
    tmp.replace(path)

def without_options(arguments, values, flags):
    result, at = [], 0
    while at < len(arguments):
        option = arguments[at]
        if option in values:
            require(at + 1 < len(arguments), 'protected_argument_missing_value')
            at += 2
        elif option in flags:
            at += 1
        else:
            result.append(option); at += 1
    return result


class CommittedRegime:
    """Read only committed native status/window rows, retaining one journal cursor."""
    DOMAINS = {'FORTS_SESSIONSTATE_REPL': ('session_state', 'sess_id', 'sessions'),
               'FORTS_INSTRUMENTSTATE_REPL': ('instrument_state', 'isin_id', 'instruments'),
               'FORTS_REFDATA_REPL': ('session', 'sess_id', 'windows')}
    def __init__(self, path):
        self.path, self.offset = path, 0
        stat = path.stat(); self.identity = stat.st_dev, stat.st_ino
        self.pending, self.begun = {}, set()
        self.sessions, self.instruments, self.windows = {}, {}, {}
        self.instrument_status_session_id = self.instrument_status_epoch = 0
        self.instrument_online = False
        self.instrument_marker_revision = -1

    def update(self):
        with self.path.open('rb') as source:
            stat = os.fstat(source.fileno())
            require((stat.st_dev, stat.st_ino) == self.identity and stat.st_size >= self.offset,
                    'native_journal_identity_or_cursor_changed')
            source.seek(self.offset)
            for line in source:
                if not line.endswith(b'\n'): break
                self.offset += len(line)
                event = json.loads(line); data = event.get('data', {}); stream = data.get('stream')
                if stream not in self.DOMAINS: continue
                table, key, domain = self.DOMAINS[stream]; target = getattr(self, domain)
                name = event.get('event')
                if name in ('listener_open', 'listener_close', 'lifenum'):
                    self.pending.pop(stream, None); self.begun.discard(stream); target.clear()
                    if stream == 'FORTS_INSTRUMENTSTATE_REPL':
                        self.instrument_status_session_id = 0; self.instrument_status_epoch += 1
                        self.instrument_online = False; self.instrument_marker_revision = -1
                elif name == 'listener_online' and stream == 'FORTS_INSTRUMENTSTATE_REPL':
                    self.instrument_online = True
                elif name == 'clear_deleted' and data.get('table') == 'sys_events' and stream == 'FORTS_INSTRUMENTSTATE_REPL':
                    floor = data.get('signed_value')
                    if type(floor) is int and self.instrument_marker_revision < floor:
                        self.instrument_status_session_id = 0
                elif name == 'clear_deleted' and data.get('table') == table:
                    floor = data.get('signed_value')
                    if type(floor) is int and floor >= 0:
                        for ident, row in list(target.items()):
                            if row.get('replRev', -1) < floor: target.pop(ident)
                elif name == 'transaction_begin':
                    self.pending[stream] = []; self.begun.add(stream)
                elif name == 'stream_row' and stream in self.begun and (data.get('table') == table or
                      stream == 'FORTS_INSTRUMENTSTATE_REPL' and data.get('table') == 'sys_events'):
                    row = decoded_fields(data)
                    row.setdefault('replRev', data.get('signed_value', -1))
                    self.pending[stream].append((data.get('table'), row))
                elif name == 'transaction_commit' and stream in self.begun:
                    for row_table, row in self.pending.pop(stream, []):
                        if stream == 'FORTS_INSTRUMENTSTATE_REPL' and row_table == 'sys_events':
                            session = row.get('sess_id', 0)
                            if session != self.instrument_status_session_id:
                                if self.instrument_status_session_id and self.instrument_online:
                                    self.instrument_status_epoch += 1
                                self.instrument_status_session_id = session
                            self.instrument_marker_revision = row.get('replRev', -1)
                            continue
                        if stream == 'FORTS_INSTRUMENTSTATE_REPL': row['_status_epoch'] = self.instrument_status_epoch
                        if row.get('replAct', 0): target.pop(row.get(key), None)
                        elif row.get(key, 0) > 0: target[row[key]] = row
                    self.begun.discard(stream)

    def continuous(self, state, targets):
        self.update()
        return (self.sessions.get(state['sess_id'], {}).get('public_state') == 1 and
                self.instrument_status_session_id == state['sess_id'] and
                all(self.instruments.get(t['isin_id'], {}).get('public_state') == 1 and
                    self.instruments[t['isin_id']].get('_status_epoch') == self.instrument_status_epoch for t in targets))

    def open_window_end(self, state, now):
        self.update(); row = self.windows.get(state['sess_id'], {})
        intervals = [(row.get('begin', 0), row.get('end', 0))]
        if row.get('mon_on') == 1: intervals.append((row.get('mon_begin', 0), row.get('mon_end', 0)))
        if row.get('eve_on') == 1: intervals.append((row.get('eve_begin', 0), row.get('eve_end', 0)))
        ends = [end for begin, end in intervals if type(begin) is int and type(end) is int and begin <= now < end]
        require(ends, 'current_committed_native_open_interval_unproved')
        return max(ends)


def decoded_fields(data):
    out = {}
    for field in data.get('fields', []):
        kind = field.get('kind', 1)
        if kind == 6:
            # Native timestamps are nanoseconds. Older journals populated only
            # unsigned seconds, so accept that representation only when the
            # native timestamp is absent/zero; never interpret nanoseconds as
            # seconds or use zero unsigned over a present native timestamp.
            timestamp = field.get('timestamp_ns', 0)
            require(type(timestamp) is int, 'native_timestamp_field_invalid')
            value = timestamp // 1_000_000_000 if timestamp else field.get('unsigned')
            require(type(value) is int, 'native_timestamp_fallback_invalid')
        else:
            value = field.get('unsigned') if kind == 2 else field.get('signed')
        out[field['name']] = value
    return out


def complete_events(path, offset):
    with Path(path).open('rb') as source:
        require(os.fstat(source.fileno()).st_size >= offset, 'native_journal_truncated')
        source.seek(offset)
        for line in source:
            if not line.endswith(b'\n'): break
            event = json.loads(line)
            require(isinstance(event, dict) and isinstance(event.get('data', {}), dict), 'native_record_invalid')
            yield event

def checkpoint_highwaters(path):
    fields = Path(path).read_text().split()
    require(len(fields) == 8 and fields[0] == 'MOEXJ2', 'durable_checkpoint_schema_unproved')
    checksum = 14695981039346656037
    for byte in ' '.join(fields[:7]).encode(): checksum = ((checksum ^ byte) * 1099511628211) & ((1 << 64) - 1)
    require(int(fields[7]) == checksum, 'durable_checkpoint_checksum_invalid')
    values = tuple(int(fields[i]) for i in (4, 5))
    require(all(x > 0 for x in values), 'durable_checkpoint_highwater_invalid')
    return values

def selective_recovery(events, after, required, kind):
    events = list(events); replication = set(required) - {'p2mqreply'}
    opened, online, reset, gaps, commits, heartbeats = set(), set(), set(), [], [], []
    for event in events:
        data = event['data']; stream = data.get('stream'); name = event['event']
        if name == 'listener_open': opened.add(stream)
        if name == 'listener_online': online.add(stream)
        if name in ('listener_close', 'listener_recovery') and stream in replication: reset.add(stream)
        if name == 'link_lost': gaps.append(data)
        if name == 'transaction_commit' and stream == 'FORTS_TRADE_REPL': commits.append(event)
        if name == 'trade_server_time' and stream == 'FORTS_TRADE_REPL' and data.get('repl_rev', 0) > 0:
            heartbeats.append(event)
    require(gaps, 'native_fault_not_observed')
    require(commits, 'postfault_trade_commit_unproved')
    if kind == 'router': require(replication <= opened & online, 'router_full_stream_recovery_unproved')
    else:
        require(reset <= opened & online, 'affected_replication_stream_recovery_unproved')
        require('p2mqreply' in opened or not any(x.get('publisher_state') != 3 for x in gaps),
                'affected_reply_listener_recovery_unproved')
    require(after['transport']['connection']['state'] == after['transport']['publisher']['state'] == 3 and
            len(after['transport']['listeners']) == len(required) and
            {r['stream'] for r in after['transport']['listeners']} == set(required) and
            all(r['native']['state'] == 3 and r['online'] is True and r['snapshot_complete'] is True and
                r['error']['code'] == 0 for r in after['transport']['listeners']) and
            not after['reconstructing'], 'current_native_listener_authority_unproved')
    return dict(observed_native_outage=True, affected_replication_stream_count=len(reset),
                retained_replication_stream_count=len(replication-reset),
                fresh_trade_commits=len(commits), compact_trade_heartbeat_samples=len(heartbeats),
                all_streams_current_native_ready=True, all_eight_fresh_commits='NOT_CLAIMED')

def load_module(name, source, expected):
    require(sha(source) == expected, 'protected_module_hash_changed')
    spec = importlib.util.spec_from_file_location(name, source)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

    def bounded_outage_probe(self, name, folder):

                'actual_fault_receipt_missing')

                'native_outage_no_longer_observed_before_probe')

        self.one_working(state)

    def working_fault(self, name, kind):
        before=self.entry_gate(); self.one_working(before)
        require(self.cfg['cleanup_deadline_utc']-time.time()>=150, 'fault_restore_and_cleanup_budget_insufficient')
        require(before['configuration']['risk']['kill_switch'] is True,'fault_requires_kill_switch_guard')
        self.passive_price=next(r for r in before['orders'] if r['client_order_id']==self.passive_cid)['price']
        # Native status prices are decimal strings; convert once without changing the original order.
        self.passive_price=self.sequence._units(self.passive_price)
        offset=(self.product.work/'interaction.ndjson').stat().st_size
        folder=self.work/name; folder.mkdir(mode=0o700)
        self.public['phase']=name; self.touch(); started=time.monotonic()
        worker=os.fork()
        if worker==0:
            # Only the parent owns the product guardian's normal-closeout pipe.
            os.close(self.guard_control)
            identity=self.product_identity; ops=self.ops
            class OwnedProcess:
                pid=identity['pid']
                def poll(child):
                    try:
                        require(ops.process_identity(child.pid)==identity,'fault_worker_product_identity_changed'); return None
                    except (FileNotFoundError,ProcessLookupError): return 0
            self.product.child=OwnedProcess()
            try:
                auth=self.fault_authorization()
                if kind=='router':
                    result=self.router.restart(self.probe,auth,self.product.child.pid,folder,
                        arm=lambda *args:self.router.arm_rollback(*args,keep_fd=self.fault_lock.fileno()))
                    witness=offset
                else:
                    witness=[None]
                    def acknowledge():
                        ready=self.wait_ready(timeout=40); self.preserve_order(before,ready)
                        witness[0]=(self.product.work/'interaction.ndjson').stat().st_size
                    result=self.upstream.interrupt(self.probe,auth,self.product.child.pid,folder,self.fault_lock.fileno(),
                                                   acknowledge=acknowledge)
                    witness=witness[0]
                atomic(folder/'helper-result-private.json',dict(result=result,witness_offset=witness))
                os._exit(0)
            except BaseException as error:
                (folder/'helper-error-private.txt').write_text(traceback.format_exc())
                atomic(folder/'helper-error-public.json',dict(error_class=type(error).__name__,fault_result='UNKNOWN'))
                os._exit(2)
        self.fault_worker=worker
        until=min(self.cfg['entry_deadline_utc'],time.time()+65); refusal=None; status=None
        try:
            while time.time()<until:
                self.touch()
                if (folder/'helper-error-public.json').exists(): raise Refused('independent_fault_helper_failed')
                if refusal is None and (folder/'fault.json').exists():
                    fault=json.loads((folder/'fault.json').read_text())
                    actual=fault.get('signal_sent') is True or fault.get('upstream_fault') is True
                    if actual:
                        state=self.product.status()
                        if state['transport']['connection']['state']!=3 or state['transport']['publisher']['state']!=3:
                            refusal=self.bounded_outage_probe(name,folder)
                done, code=os.waitpid(worker,os.WNOHANG)
                if done: status=code; self.fault_worker=None; break
                time.sleep(.02)
            require(status==0 and refusal is not None,'fault_or_single_refusal_not_completed_within_bound')
        finally:
            self.wait_fault_worker()
        saved=json.loads((folder/'helper-result-private.json').read_text()); after=self.wait_ready()
        self.one_working(after); self.preserve_order(before,after)
        proof_until=min(self.cfg['entry_deadline_utc'],time.time()+10)
        while True:
            events=list(complete_events(self.product.work/'interaction.ndjson',saved['witness_offset']))
            if any(e['event']=='transaction_commit' and e['data'].get('stream')=='FORTS_TRADE_REPL' for e in events): break
            require(time.time()<proof_until,'postfault_trade_commit_deadline')
            self.touch(); time.sleep(.05)
        witness=selective_recovery(events,after,self.ops.REQUIRED,kind)
        self.public['cases'].append(dict(case=name,status='PASS',working_order_preserved=True,
            local_refusal=refusal,recovery=saved['result'],native_witness=witness,
            duration_ms=round((time.monotonic()-started)*1000),ready_utc_ns=time.time_ns()))
        self.touch()

    def journal_counts(self, offset):
        counts = {'AddOrder': 0, 'MoveOrder': 0}
        with (self.product.work / 'interaction.ndjson').open('rb') as stream:
            stream.seek(offset)

                if event.get('event') == 'command' and event['data'].get('name') in counts:

    def preserve_order(self, before, after):
        original = [r for r in before['orders'] if r['state'] not in TERMINAL]
        current = [r for r in after['orders'] if r['state'] not in TERMINAL]
        require(len(original) == len(current), 'working_order_population_changed')

                       all(r[k] == row[k] for k in ('order_id', 'ext_id', 'isin_id', 'quantity',
                                                   'executed', 'remaining', 'price'))]
            require(len(matches) == 1, 'working_order_or_fill_baseline_changed')
        require(after['positions'] == before['positions'], 'position_baseline_changed')

    def arm_product_guard(self):
        require(self.guardian is None, 'previous_product_guardian_not_closed')
        ready_r, ready_w = os.pipe(); control_r, control_w = os.pipe()
        child = os.fork()
        if child == 0:
            os.close(ready_r); os.close(control_w); os.setsid()
            with open(os.devnull, 'r+b', buffering=0) as null:
                for fd in (0, 1, 2):
                    os.dup2(null.fileno(), fd)
            try:
                os.write(ready_w, b'ARMED'); os.close(ready_w)
                # Ordinary completion sends D. EOF or a stale/bounded lease
                # performs actual CLI risk-reducing cleanup on this owned PID.
                import select
                while time.time() < self.cfg['cleanup_deadline_utc']:
                    readable, _, _ = select.select([control_r], [], [], 1)
                    if readable:
                        if os.read(control_r, 1) == b'D':
                            atomic(self.product.work / 'guardian-done.json', dict(state='NORMAL_PARENT_CLOSEOUT',
                                                                                no_guardian_failure=True))
                            os._exit(0)
                        break
                    receipt = json.loads((self.work / 'public.json').read_text())
                    if time.time_ns() - receipt.get('updated_utc_ns', 0) > 120_000_000_000:
                        break
                self.guardian_cleanup()
                os._exit(0)
            except BaseException as error:
                atomic(self.product.work / 'guardian-done.json',
                       dict(state='BLOCKED', error_class=type(error).__name__, flatness='UNKNOWN'))
                os._exit(2)
        os.close(ready_w); os.close(control_r)
        import select
        readable, _, _ = select.select([ready_r], [], [], 2)
        require(readable and os.read(ready_r, 5) == b'ARMED', 'product_guardian_not_armed')
        os.close(ready_r)
        self.guardian, self.guard_control = child, control_w
        atomic(self.product.work / 'guardian-owned-private.json', self.ops.process_identity(child))

    def guardian_cleanup(self):
        product, identity, ops = self.product, self.product_identity, self.ops
        (product.work / 'stop-requested').touch()
        # A forked guardian cannot waitpid() the CLI. Its native process proof
        # uses exact /proc identity; it never invents a process exit code.
        class OwnedProcess:
            pid = identity['pid']
            def poll(child):
                try:
                    require(ops.process_identity(child.pid) == identity, 'guardian_product_identity_changed')
                    return None
                except (FileNotFoundError, ProcessLookupError):
                    return 0
        product.child = OwnedProcess(); product.risk_guard = None
        product.touch = lambda: None
        # Router helpers independently restore their exact recipe. Wait only
        # read-only status until full native reconstruction is current again.
        until = min(self.cfg['cleanup_deadline_utc'], time.time() + 600)
        state = product.wait_ready(until, entry=False)
        self.inspect(); self.configuration(state)
        result = (ops.clean_product(product, until=until, final=True) if self.allow_orders else
                  dict(cancel_confirmed=ops.flat(state), flat=ops.flat(state),
                       cancellation='not_required_complete_native_empty_snapshot'))
        require(result['cancel_confirmed'] is True and result['flat'] is True,
                'guardian_cancellation_or_flatness_unproved')
        self.full_state(flat=True); self.inspect()
        require(ops.checked_signal(identity, signal.SIGTERM), 'guardian_owned_sigterm_not_delivered')
        end = time.monotonic() + 20
        while product.child.poll() is None and time.monotonic() < end:
            time.sleep(.1)
        require(product.child.poll() is not None, 'guardian_product_did_not_exit')
        atomic(product.work / 'guardian-done.json',
               dict(state='RISK_REDUCED', cancel_confirmed=True, flat=True, process_absent=True,
                    product_exit='UNKNOWN_GUARDIAN_CANNOT_WAITPID', no_guardian_failure=True))

    def cleanup(self):
        if self.product is None or self.product.child.poll() is not None:
            return None
        self.product.risk_guard = None
        self.public['phase'] = 'cleanup'; self.touch()
        self.wait_fault_worker()
        # Existing bounded cleanup preserves original quantity/position/caps,
        # does not retry uncertain Adds and uses the actual command socket.
        self.wait_ready()
        result = (self.ops.clean_product(self.product, until=min(self.cfg['cleanup_deadline_utc'],
                                                                time.time() + 180), final=True)
                  if self.allow_orders else dict(cancel_confirmed=True, flat=True,
                                                 cancellation='not_required_complete_native_empty_snapshot'))
        if not self.allow_orders:
            self.full_state(flat=True)
        require(result['cancel_confirmed'] is True and result['flat'] is True, 'cleanup_flatness_unproved')
        self.full_state(flat=True)
        self.product.close()  # Owned SIGTERM only; never SIGKILL or a second signal.
        result['product_exit'] = self.product.child.returncode
        require(result['product_exit'] == 0, 'normal_cli_exit_not_zero')
        self.finish_guard(self.product.work)
        label = self.product.work.name
        self.product = None
        self.no_clients()
        result['independent_after_flatness'] = self.independent_flat(label + '-after-flatness')
        return result

        self.sequence = load_module('sequence', q / self.cfg['sequence_relative'],
                                    self.cfg['protected_files'][self.cfg['sequence_relative']])
        self.coverage = load_module('interruption_coverage', q / self.cfg['coverage_relative'],
                                    self.cfg['protected_files'][self.cfg['coverage_relative']])
        require(sha(Path('/proc/sys/kernel/random/boot_id')) == self.cfg['boot_sha256'], 'boot_changed')
        self.full_day_closeout_gate(q)
        folder = self.work / 'native-env'; folder.mkdir(mode=0o700)
        base_cfg = json.loads((q / self.cfg['bootstrap_inputs_relative']).read_text())
        self.environment = bootstrap.setup(folder, base_cfg)
        self.live_checkpoint_before = sha(self.environment['state'])
        self.initial_highwaters = checkpoint_highwaters(self.environment['state'])
        require(self.environment['inventory']()['client_count'] == 0, 'existing_router_client')
        cells = [c.cell_contents for c in self.environment['inventory'].__closure__]
        self.probe = next(c for c in cells if hasattr(c, 'inspect') and hasattr(c, 'clients'))
        self.original_router, _ = self.probe.inspect(dict(environment='test'))
        white = self.upstream.pristine_whitelist(self.probe, self.original_router, T1_LINKS_SHA)
        self.initial_peers = {r.peer for r in self.upstream.upstreams(self.probe, self.original_router.pid, white)}
        self.fault_lock = (helpers / 'router-fault.lock').open('a')
        fcntl.flock(self.fault_lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        begin, end = map(int, base_cfg['ext_id_range'].split(':'))
        self.budget = lambda: self.ops.capacity(self.environment['state'], q / self.cfg['allocation_receipt_relative'],
                                               base_cfg['instance'], begin, end)
        self.budget()
        resolver = q / self.cfg['resolver_relative']
        require(sha(resolver) == self.cfg['protected_files'][self.cfg['resolver_relative']], 'resolver_changed')
        with (self.work / 'refdata-private.json').open('x') as out, (self.work / 'refdata-error-private.log').open('x') as err:
            result = subprocess.Popen([str(resolver), *self.environment['common'], '--read-only-market-data',
                                       '--isin-id', '1', '--wait-ms', '60000'], cwd=folder,
                                      env=self.environment['env'], stdout=out, stderr=err, stdin=subprocess.DEVNULL)
            identity = self.ops.process_identity(result.pid)
            try:
                result.wait(timeout=80)
            except subprocess.TimeoutExpired:
                self.ops.checked_signal(identity, signal.SIGTERM)
                result.wait(timeout=20)
                raise Refused('fresh_refdata_resolver_deadline')
        require(result.returncode == 0, 'fresh_committed_refdata_resolution_failed')

            target.update(quote_cap_scaled=cap, quote_reference_scaled=cap // 2)
        self.no_clients()
        common = without_options(self.environment['common'],
                                 {'--max-commands-per-second', '--max-quantity', '--max-open-orders',
                                  '--login-env', '--ext-id-range', '--state'}, {'--allow-orders', '--sole-instance'})

                              '--max-position', str(target['isin_id']) + '=2']
        ops = self.ops; controller = self
        # This newly authorized session owns its explicit same-day deadline;
        # no executed script/bundle or historical receipt is changed.
        ops.CLEANUP = self.cfg['entry_deadline_utc']
        self.ops.CLEANUP = self.cfg['entry_deadline_utc']
        class ManagedProduct(ops.Product):
            def close(product):
                if product.child is not None and product.child.poll() is None:
                    controller.inspect()
                    require(not getattr(product, 'sigterm_sent', False), 'owned_sigterm_already_sent')
                    require(ops.checked_signal(controller.product_identity, signal.SIGTERM), 'owned_sigterm_not_delivered')
                    product.sigterm_sent = True
                    product.child.wait(timeout=20)
        self.Product = ManagedProduct
        self.public.update(state='PREFLIGHT_PASS', timer_paused=True, after=self.no_clients(),
                           remaining_ids=self.budget(), trading_armed=False, authorized_date=self.cfg['authorized_date'])
        self.touch()


def run_after_full_day(cfg, work):
    """Explicit executor entry point. Root seals/installs cfg and work before calling."""
    sys.dont_write_bytecode = True; os.umask(0o077)
    authorized(cfg)
    work = Path(work).resolve(strict=True)
    require(not (work / 'public.json').exists(), 'one_shot_work_already_used')
    Controller(work, cfg).execute()
    return json.loads((work / 'public.json').read_text())


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--config', type=Path, required=True)
    parser.add_argument('--work', type=Path, required=True)
    parser.add_argument('--execute', action='store_true')
    args = parser.parse_args()
    sys.dont_write_bytecode = True; os.umask(0o077)
    cfg = json.loads(args.config.read_text())
    authorized(cfg)
    require(args.execute, 'preparation_only_native_execution_not_selected')
    # The controller and its independent guardians live on TEST after SSH loss.
    signal.signal(signal.SIGHUP, signal.SIG_IGN)
    try:
        os.setsid()
    except PermissionError:
        require(os.getsid(0) == os.getpid(), 'controller_session_detachment_unproved')
    work = args.work.resolve(strict=True)
    require(not (work / 'public.json').exists(), 'one_shot_work_already_used')
    run_after_full_day(cfg, work)


if __name__ == '__main__':
    main()
